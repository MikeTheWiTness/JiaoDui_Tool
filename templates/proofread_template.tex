% === 内部辅助：计算 paracol 列宽 ===
% paracol 中 \linewidth = 列宽，减掉 fbox 边框和间距
\newlength{\colwidthinner}
\newcommand{\calccolwidth}{%
  \setlength{\colwidthinner}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}%
}

% === 自定义命令：修改标记 ===
% text 类型：红色删除线（CJK 兼容）+ 蓝色正确文字
\newcommand{\corrtext}[2]{%
  \textcolor{red}{\CJKsout{#1}}~\textcolor{blue}{#2}%
}

% rewrite 类型：红色框包裹原文（自动换行）
\newcommand{\corrrewrite}[1]{%
  \calccolwidth
  \fcolorbox{red}{red!10}{\parbox{\colwidthinner}{#1}}%
}

% region 类型：红色框标记区域（自动换行）
\newcommand{\corrregion}[1]{%
  \calccolwidth
  \fcolorbox{red}{red!10}{\parbox{\colwidthinner}{#1}}%
}

% 修改建议高亮框（右侧栏）
\newcommand{\correctionbox}[1]{%
  \calccolwidth
  \colorbox{blue!8}{\parbox{\colwidthinner}{#1}}%
}
